# Flutter UI Fundamentals â€” Course Explorer

Aplikasi Course Explorer dibuat menggunakan Flutter untuk menampilkan daftar mata kuliah, detail mata kuliah, status penyelesaian, dan fitur favorit. Aplikasi juga menampilkan identitas mahasiswa serta mendukung navigasi yang adaptif.

## Struktur Folder dan Tanggung Jawab

* **`lib/models/`**
  Menyimpan model data, seperti `Course`, untuk merepresentasikan informasi mata kuliah.

* **`lib/providers/`**
  Mengelola state aplikasi, termasuk daftar mata kuliah, status loading, pesan error, dan daftar favorit.

* **`lib/repositories/`**
  Menjadi perantara antara Provider dan Service dalam pengambilan data.

* **`lib/services/`**
  Menangani sumber data, termasuk membaca file JSON dari assets dan mengubahnya menjadi objek yang dapat digunakan aplikasi.

* **`lib/screens/`**
  Menyimpan halaman aplikasi, seperti Home, Courses, Favorites, Profile, dan Detail Course.

* **`lib/widgets/`**
  Menyimpan widget yang dapat digunakan kembali, seperti `CourseCard` dan `StudentHeader`.

* **`lib/constants.dart`**
  Menyimpan konstanta yang digunakan bersama, seperti nama dan NIM mahasiswa.

* **`lib/course_state.dart`**
  Jika file ini masih digunakan, jelaskan tanggung jawab sebenarnya berdasarkan isi kodenya. Jika tidak digunakan, periksa dahulu sebelum menghapusnya.

* **`lib/main.dart`**
  Menjadi titik masuk aplikasi, mengatur konfigurasi `MaterialApp`, Provider, serta navigasi dan layout utama.

* **`assets/data/`**
  Menyimpan data JSON mahasiswa dan mata kuliah yang digunakan aplikasi.

## Arah Dependency

Arah dependency yang diterapkan untuk pengambilan data mata kuliah adalah:

`Screen/Widget â†’ Provider â†’ Repository â†’ Service/Data Source`

Screen dan widget menampilkan data serta menerima interaksi pengguna. Provider mengelola state, Repository menjadi perantara pengambilan data, sedangkan Service menangani akses ke sumber data.

UI tidak membaca JSON secara langsung menggunakan `rootBundle` atau `jsonDecode`.

import 'package:flutter/foundation.dart';

import '../models/course.dart';
import '../repositories/course_repository.dart';

class CourseProvider extends ChangeNotifier {
  final CourseRepository repository;

  CourseProvider(this.repository);

  List<Course> courses = [];

  bool isLoading = false;
  bool isSaving = false;
  bool isDeleting = false;

  String? error;
  String? refreshError;

  final Set<String> favorites = {};

  bool isFavorite(String code) {
    return favorites.contains(code);
  }

  void toggleFavorite(String code) {
    if (favorites.contains(code)) {
      favorites.remove(code);
    } else {
      favorites.add(code);
    }

    notifyListeners();
  }

  List<Course> get favoriteCourses {
    return courses
        .where((course) => favorites.contains(course.code))
        .toList();
  }

  Future<void> loadCourses() async {
    isLoading = true;
    error = null;
    refreshError = null;

    notifyListeners();

    try {
      final loadedCourses = await repository.getCourses();
      courses = loadedCourses;
    } catch (e) {
      error = e.toString();
    } finally {
      isLoading = false;
      notifyListeners();
    }
  }

  Future<Course> createCourse(Course course) async {
    if (isSaving || isDeleting) {
      throw Exception('Operasi course sedang berlangsung.');
    }

    isSaving = true;
    error = null;
    refreshError = null;

    notifyListeners();

    try {
      final createdCourse = await repository.createCourse(course);

      try {
        final loadedCourses = await repository.getCourses();
        courses = loadedCourses;
      } catch (e) {
        refreshError = e.toString();
      }

      return createdCourse;
    } catch (e) {
      error = e.toString();
      rethrow;
    } finally {
      isSaving = false;
      notifyListeners();
    }
  }

  Future<Course> updateCourse(int id, Course course) async {
    if (isSaving || isDeleting) {
      throw Exception('Operasi course sedang berlangsung.');
    }

    isSaving = true;
    error = null;
    refreshError = null;

    notifyListeners();

    try {
      final updatedCourse =
          await repository.updateCourse(id, course);

      try {
        final loadedCourses = await repository.getCourses();
        courses = loadedCourses;
      } catch (e) {
        refreshError = e.toString();
      }

      return updatedCourse;
    } catch (e) {
      error = e.toString();
      rethrow;
    } finally {
      isSaving = false;
      notifyListeners();
    }
  }

  Future<void> deleteCourse(int id) async {
    if (isSaving || isDeleting) {
      throw Exception('Operasi course sedang berlangsung.');
    }

    final courseIndex = courses.indexWhere(
      (course) => course.id == id,
    );

    if (courseIndex == -1) {
      throw Exception('Course tidak ditemukan.');
    }

    final course = courses[courseIndex];

    isDeleting = true;
    error = null;
    refreshError = null;

    notifyListeners();

    try {
      // 1. Kirim permintaan DELETE ke API.
      await repository.deleteCourse(id);

      // 2. Hapus course dari daftar lokal.
      courses.removeWhere((item) => item.id == id);

      // 3. Hapus juga favorite berdasarkan kode course.
      favorites.remove(course.code);

      // 4. Coba sinkronkan ulang daftar dari API.
      // Kegagalan GET tidak membatalkan DELETE yang berhasil.
      try {
        final loadedCourses = await repository.getCourses();
        courses = loadedCourses;

        // Bersihkan favorite yang tidak lagi ada di daftar.
        final validCodes = courses
            .map((item) => item.code)
            .toSet();

        favorites.removeWhere(
          (code) => !validCodes.contains(code),
        );
      } catch (e) {
        refreshError = e.toString();
      }
    } catch (e) {
      error = e.toString();
      rethrow;
    } finally {
      isDeleting = false;
      notifyListeners();
    }
  }
}

class CourseFormPage extends StatefulWidget {

class _CourseFormPageState extends State<CourseFormPage> {
  final _formKey = GlobalKey<FormState>();

  final _codeController = TextEditingController();
  final _titleController = TextEditingController();
  final _creditsController = TextEditingController();

  String _status = 'planned';
  bool _isSubmitting = false;

  @override
  void initState() {
    super.initState();

    // Isi form dengan data lama jika sedang mengedit.
    final course = widget.course;

    if (course != null) {
      _codeController.text = course.code;
      _titleController.text = course.title;
      _creditsController.text = course.credits.toString();
      _status = course.status;
    }
  }

  @override
  void dispose() {
    _codeController.dispose();
    _titleController.dispose();
    _creditsController.dispose();
    super.dispose();
  }

  Future<void> _saveCourse() async {
    if (_isSubmitting) return;

    if (!_formKey.currentState!.validate()) {
      return;
    }

    final provider = context.read<CourseProvider>();
    final messenger = ScaffoldMessenger.of(context);
    final navigator = Navigator.of(context);

    final existingCourse = widget.course;

    // Saat edit, ID lama harus dipertahankan.
    if (existingCourse != null && existingCourse.id == null) {
      messenger.showSnackBar(
        const SnackBar(
          content: Text('Course tidak memiliki ID untuk diperbarui.'),
          backgroundColor: Colors.red,
        ),
      );
      return;
    }

    final course = Course(
      id: existingCourse?.id,
      code: _codeController.text.trim(),
      title: _titleController.text.trim(),
      credits: int.parse(_creditsController.text.trim()),
      status: _status,
    );

    setState(() {
      _isSubmitting = true;
    });

    try {
      if (widget.isEditing) {
        await provider.updateCourse(
          existingCourse!.id!,
          course,
        );
      } else {
        await provider.createCourse(course);
      }

      if (!mounted) return;

      messenger.showSnackBar(
        SnackBar(
          content: Text(
            widget.isEditing
                ? 'Permintaan update course berhasil.'
                : 'Course berhasil dibuat.',
          ),
          backgroundColor: Colors.green,
        ),
      );

      navigator.pop();
    } catch (e) {
      if (!mounted) return;

      messenger.showSnackBar(
        SnackBar(
          content: Text(
            '${widget.isEditing ? 'Gagal memperbarui' : 'Gagal membuat'} '
            'course: $e',
          ),
          backgroundColor: Colors.red,
        ),
      );
    } finally {
      if (mounted) {
        setState(() {
          _isSubmitting = false;
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final provider = context.watch<CourseProvider>();
    final isSaving = _isSubmitting || provider.isSaving;

    return Scaffold(

        ),
        centerTitle: true,
      ),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(20),
        child: Form(
          key: _formKey,
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.stretch,
            children: [
              TextFormField(
                controller: _codeController,
                readOnly: widget.isEditing,
                decoration: InputDecoration(
                  labelText: 'Code',
                  hintText: 'Contoh: PTI101',
                  border: const OutlineInputBorder(),
                  helperText: widget.isEditing
                      ? 'Code tidak dapat diubah saat edit.'

                ),

class CourseApiService implements CourseDataSource {

  CourseApiService(this.baseUrl);

  @override
  Future<List<Course>> getCourses() async {
    final uri = Uri.parse(baseUrl);

    debugPrint('GET URL: $uri');

    final response = await http.get(uri);

    debugPrint('GET STATUS: ${response.statusCode}');

    if (response.statusCode != 200) {
      throw Exception(
        'Gagal mengambil data course: '
        'HTTP ${response.statusCode}. '
        'Respons: ${response.body}',
      );
    }

    final List<dynamic> data =
        jsonDecode(response.body) as List<dynamic>;

    return data.map((item) {
      final post = item as Map<String, dynamic>;

      return Course(
        id: post['id'] as int?,
        code: 'POST-${post['userId']}',

        // KASUS C â€“ JSON MAPPING ERROR
        // Nama key sengaja dibuat salah.
        // Kode ini menyebabkan kesalahan saat title dibaca.
        title: post['title_wrong'] as String,

        credits: 0,
        status: 'planned',
      );
    }).toList();
  }

  @override
  Future<Course> createCourse(Course course) async {
    final uri = Uri.parse(baseUrl);

    debugPrint('POST URL: $uri');

    final response = await http.post(
      uri,
      headers: {
        'Content-Type': 'application/json',
      },
      body: jsonEncode(course.toJson()),
    );

    debugPrint('POST STATUS: ${response.statusCode}');
    debugPrint('POST RESPONSE: ${response.body}');

    if (response.statusCode == 200 ||
        response.statusCode == 201) {
      return Course.fromJson(
        jsonDecode(response.body) as Map<String, dynamic>,
      );
    }

    throw Exception(
      'Gagal menambahkan course: '
      'HTTP ${response.statusCode}. '
      'Respons: ${response.body}',
    );
  }

  @override
  Future<Course> updateCourse(int id, Course course) async {
    final uri = Uri.parse('$baseUrl/$id');

    debugPrint('PUT URL: $uri');

    final response = await http.put(
      uri,
      headers: {
        'Content-Type': 'application/json',
      },
      body: jsonEncode(course.toJson()),
    );

    debugPrint('PUT STATUS: ${response.statusCode}');
    debugPrint('PUT RESPONSE: ${response.body}');

    if (response.statusCode >= 200 &&
        response.statusCode < 300) {
      return Course.fromJson(
        jsonDecode(response.body) as Map<String, dynamic>,
      );
    }

    throw Exception(
      'Gagal memperbarui course: '
      'HTTP ${response.statusCode}. '
      'Respons: ${response.body}',
    );
  }

  @override

    final uri = Uri.parse('$baseUrl/$id');

    debugPrint('DELETE URL: $uri');

    debugPrint('DELETE STATUS: ${response.statusCode}');
    debugPrint('DELETE RESPONSE: ${response.body}');

  }

import 'package:supabase_flutter/supabase_flutter.dart';

class CourseSupabaseService implements CourseDataSource {
  final supabase = Supabase.instance.client;

  @override
  Future<List<Course>> getCourses() async {
    final data = await supabase
        .from('courses')
        .select()
        .order('id');

    return (data as List)
        .map((e) => Course.fromJson(e as Map<String, dynamic>))
        .toList();
  }

  @override
  Future<Course> createCourse(Course course) async {
    final userId = supabase.auth.currentUser?.id;

    if (userId == null) {
      throw Exception('Pengguna belum login.');
    }

    final payload = course.toJson()
      ..remove('id');

    // Mengisi owner_id dengan ID pengguna yang sedang login.
    payload['owner_id'] = userId;

    final row = await supabase
        .from('courses')
        .insert(payload)
        .select()
        .single();

    return Course.fromJson(row);
  }

  @override
  Future<Course> updateCourse(int id, Course course) async {
    final payload = course.toJson()
      ..remove('id');

    final row = await supabase
        .from('courses')
        .update(payload)
        .eq('id', id)
        .select()
        .single();

    return Course.fromJson(row);
  }

  @override
  Future<void> deleteCourse(int id) async {
    await supabase
        .from('courses')
        .delete()
        .eq('id', id);
  }
}
